cmake_minimum_required(VERSION 3.20)
project(JoltFXSDKJoltscriptExample LANGUAGES C CXX)
find_package(JoltFXSDK 1.1 REQUIRED CONFIG COMPONENTS joltscript)
find_package(JoltFX REQUIRED CONFIG)
add_executable(sdk_joltscript main.c)
target_link_libraries(sdk_joltscript PRIVATE JoltFXSDK::joltscript JoltFX::joltscript_glue)
set_target_properties(sdk_joltscript PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
enable_testing()
add_test(NAME sdk_joltscript_typed COMMAND sdk_joltscript)
