cmake_minimum_required(VERSION 3.20)
project(JoltFXSDKIngagiExample LANGUAGES C)
find_package(JoltFXSDK 1.1 REQUIRED CONFIG COMPONENTS ingagi)
find_library(INGAGI_EMBED_LIBRARY NAMES ingagi_embed REQUIRED)
add_executable(sdk_ingagi main.c)
target_link_libraries(sdk_ingagi PRIVATE JoltFXSDK::ingagi "${INGAGI_EMBED_LIBRARY}")
set_target_properties(sdk_ingagi PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
enable_testing()
add_test(NAME sdk_ingagi_scalar COMMAND sdk_ingagi)
