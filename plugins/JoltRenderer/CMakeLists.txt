cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(JoltRenderer VERSION 1.0.0 LANGUAGES C CXX)
    find_package(JoltFXSDK 1 REQUIRED CONFIG)
endif()
include(GNUInstallDirs)
option(JFX_JOLT_RENDERER_SIMD "Use xsimd for Jolt Renderer's pixel resolve" ON)
set(JFX_JOLT_RENDERER_XSIMD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/xsimd" CACHE PATH "xsimd source root")
function(jolt_renderer_settings target)
    target_compile_features(${target} PRIVATE cxx_std_17)
    if(JFX_JOLT_RENDERER_SIMD AND EXISTS "${JFX_JOLT_RENDERER_XSIMD_ROOT}/include/xsimd/xsimd.hpp")
        target_include_directories(${target} SYSTEM PRIVATE "${JFX_JOLT_RENDERER_XSIMD_ROOT}/include")
        target_compile_definitions(${target} PRIVATE JFX_JOLT_RENDERER_XSIMD=1)
    endif()
endfunction()
# Browser/WASI/iOS hosts use static attachment, never native module loading.
if(NOT EMSCRIPTEN AND NOT CMAKE_SYSTEM_NAME MATCHES "^(iOS|tvOS|watchOS|WASI)$")
    joltfx_sdk_add_renderer(JoltRenderer src/jolt_renderer.cpp)
    target_compile_definitions(JoltRenderer PRIVATE JFX_JOLT_RENDERER_MODULE=1)
    jolt_renderer_settings(JoltRenderer)
    install(TARGETS JoltRenderer LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/joltfx/renderers
        RUNTIME DESTINATION ${CMAKE_INSTALL_LIBDIR}/joltfx/renderers)
endif()
if(TARGET jfx_core)
    add_library(jfx_jolt_renderer STATIC src/jolt_renderer.cpp)
    target_link_libraries(jfx_jolt_renderer PRIVATE JoltFXSDK::renderer)
    set_target_properties(jfx_jolt_renderer PROPERTIES POSITION_INDEPENDENT_CODE ON)
    jolt_renderer_settings(jfx_jolt_renderer)
    target_link_libraries(jfx_core PRIVATE jfx_jolt_renderer JoltFXSDK::renderer)
    target_compile_definitions(jfx_core PUBLIC JFX_HAVE_JOLT_RENDERER=1)
endif()
