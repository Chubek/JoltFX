cmake_minimum_required(VERSION 3.20)
project(JoltFXScriptHost LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
find_package(JoltFX CONFIG REQUIRED)
add_executable(script_host main.c)
target_link_libraries(script_host PRIVATE JoltFX::jfx_extif)
option(JFX_EXAMPLE_SANITIZERS "Link against a sanitizer-instrumented JoltFX installation" OFF)
if(JFX_EXAMPLE_SANITIZERS)
    target_compile_options(script_host PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(script_host PRIVATE -fsanitize=address,undefined)
endif()
