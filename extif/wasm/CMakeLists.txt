set(JFX_WAMR_ROOT "${PROJECT_SOURCE_DIR}/third_party/wasm-micro-runtime" CACHE PATH
    "WebAssembly Micro Runtime source checkout")
if(NOT EXISTS "${JFX_WAMR_ROOT}/build-scripts/runtime_lib.cmake")
    message(FATAL_ERROR "JFX_EXT_WASM requires third_party/wasm-micro-runtime. Run scripts/submodules-init.sh or set JFX_WAMR_ROOT to a WAMR checkout.")
endif()
file(READ "${JFX_WAMR_ROOT}/core/iwasm/include/wasm_export.h" wamr_header)
if(NOT wamr_header MATCHES "wasm_runtime_set_instruction_count_limit")
    message(FATAL_ERROR "JFX_WAMR_ROOT must provide WAMR instruction metering support.")
endif()
add_subdirectory(runtime)
add_library(jfx_wasm STATIC src/wasm_runtime.c)
target_include_directories(jfx_wasm PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(jfx_wasm PRIVATE ../common/src)
target_link_libraries(jfx_wasm PUBLIC jfx_extif_common PRIVATE jfx_wamr_runtime)
set_project_warnings(jfx_wasm)
