set(JFX_MRUBY_ROOT "${PROJECT_SOURCE_DIR}/third_party/mruby" CACHE PATH
    "Path to a compatible mruby source checkout (tested with 3.3.0)")
if(NOT EXISTS "${JFX_MRUBY_ROOT}/include/mruby.h")
    message(FATAL_ERROR "mruby headers not found at '${JFX_MRUBY_ROOT}'. "
        "Set -DJFX_MRUBY_ROOT=<mruby-3.3.0-checkout>.")
endif()
file(STRINGS "${JFX_MRUBY_ROOT}/include/mruby.h" JFX_MRUBY_ALLOCATOR_API
    REGEX "mrb_open_allocf" LIMIT_COUNT 1)
if(NOT JFX_MRUBY_ALLOCATOR_API)
    message(FATAL_ERROR "This mruby checkout lacks mrb_open_allocf, required "
        "for JoltFX's sandbox memory budgets. Use mruby 3.3.0 via "
        "-DJFX_MRUBY_ROOT=<checkout>, or disable mruby with -DJFX_EXT_MRUBY=OFF.")
endif()
string(SHA256 JFX_MRUBY_SOURCE_KEY "${JFX_MRUBY_ROOT}")
string(SUBSTRING "${JFX_MRUBY_SOURCE_KEY}" 0 12 JFX_MRUBY_SOURCE_KEY)
set(JFX_MRUBY_BUILD_DIR "${CMAKE_CURRENT_BINARY_DIR}/runtime/${JFX_MRUBY_SOURCE_KEY}")
set(JFX_MRUBY_LIBRARY "${JFX_MRUBY_BUILD_DIR}/jfx/lib/libmruby.a")
# Rake writes its lockfile beside the configuration. Keep both in this build
# tree, and invalidate the runtime archive when the source checkout changes.
configure_file(build_config.rb "${JFX_MRUBY_BUILD_DIR}/build_config.rb" COPYONLY)
set(JFX_MRUBY_CFLAGS "-fPIC")
set(JFX_MRUBY_WARNING_FLAGS "")
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    # Upstream's deliberate one-byte strncat in its lexer trips GCC's heuristic.
    set(JFX_MRUBY_WARNING_FLAGS "-Wno-stringop-truncation")
endif()
if(JFX_ASAN)
    string(APPEND JFX_MRUBY_CFLAGS " -fsanitize=address -fno-omit-frame-pointer")
endif()
if(JFX_UBSAN)
    string(APPEND JFX_MRUBY_CFLAGS " -fsanitize=undefined")
endif()
add_custom_command(
    OUTPUT "${JFX_MRUBY_LIBRARY}"
    COMMAND ${CMAKE_COMMAND} -E env MRUBY_CONFIG=${JFX_MRUBY_BUILD_DIR}/build_config.rb
            MRUBY_BUILD_DIR=${JFX_MRUBY_BUILD_DIR} CC=${CMAKE_C_COMPILER} AR=${CMAKE_AR}
            "JFX_MRUBY_CFLAGS=${JFX_MRUBY_CFLAGS}"
            "JFX_MRUBY_WARNING_FLAGS=${JFX_MRUBY_WARNING_FLAGS}"
            ${CMAKE_COMMAND} -E chdir ${JFX_MRUBY_ROOT} ruby minirake
    DEPENDS "${JFX_MRUBY_BUILD_DIR}/build_config.rb"
            "${JFX_MRUBY_ROOT}/include/mruby.h"
    WORKING_DIRECTORY "${JFX_MRUBY_ROOT}"
    COMMENT "Building the vendored mruby runtime"
)
add_custom_target(jfx_mruby_runtime_build DEPENDS "${JFX_MRUBY_LIBRARY}")
include(GNUInstallDirs)
set(JFX_MRUBY_INSTALLED_ARCHIVE "${CMAKE_STATIC_LIBRARY_PREFIX}jfx_mruby_runtime${CMAKE_STATIC_LIBRARY_SUFFIX}")
add_library(jfx_mruby_runtime INTERFACE)
target_link_libraries(jfx_mruby_runtime INTERFACE
    "$<BUILD_INTERFACE:${JFX_MRUBY_LIBRARY}>"
    "$<INSTALL_INTERFACE:$<INSTALL_PREFIX>/${CMAKE_INSTALL_LIBDIR}/${JFX_MRUBY_INSTALLED_ARCHIVE}>")
add_dependencies(jfx_mruby_runtime jfx_mruby_runtime_build)
install(FILES "${JFX_MRUBY_LIBRARY}" DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RENAME "${JFX_MRUBY_INSTALLED_ARCHIVE}")

add_library(jfx_mruby STATIC src/mruby_runtime.c)
target_include_directories(jfx_mruby PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
)
# Presymbol IDs are generated by the jfx cross build and must match the
# runtime archive, rather than the host mrbc compiler's symbol table.
target_include_directories(jfx_mruby SYSTEM PRIVATE
    "${JFX_MRUBY_BUILD_DIR}/jfx/include"
    "${JFX_MRUBY_ROOT}/include")
target_include_directories(jfx_mruby PRIVATE ../common/src)
target_compile_definitions(jfx_mruby PRIVATE MRB_USE_DEBUG_HOOK MRB_INT64 MRB_WORDBOX_NO_FLOAT_TRUNCATE)
target_link_libraries(jfx_mruby PUBLIC jfx_extif_common PRIVATE jfx_mruby_runtime tilly)
if(NOT MSVC)
    target_link_libraries(jfx_mruby PRIVATE m)
endif()
set_project_warnings(jfx_mruby)
